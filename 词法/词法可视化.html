-- 词频统计：读一个文本文件，统计每个词出现几次，打印前 10 名
-- 这一份既是示例程序，也是词法分析器的测试用例

模块 词频

引入 std.io
引入 std.string
引入 std.array
引入 std.map
引入 std.os
引入 std.fs

结构 词频项 {
    词: 字符串
    次数: 整数
}

-- 把文本切成词，数一数每个词出现几次
函数 统计(文本: 借用 字符串) -> 映射[字符串, 整数] {
    私有 映射[字符串, 整数] 表 = {}
    私有 数组[字符串] 词们 = std.string.分割(文本, " ")

    对 词 在 词们 {
        私有 字符串 干净 = std.string.去空白(std.string.小写(词))
        如果 #干净 == 0 就 { 继续 }

        私有 可空[整数] 旧 = 表.取(干净)
        如果 旧 == 空 就 {
            表.插入(干净, 1)
        } 否则 {
            表.插入(干净, 旧.取值() + 1)
        }
    }
    返回 表
}

-- 按次数从多到少排个榜
函数 排行(表: 借用 映射[字符串, 整数]) -> 数组[词频项] {
    私有 数组[词频项] 项们 = {}
    对 词, 次数 在 表 {
        项们.追加({ 复制 词, 次数 })
    }
    项们.排序按(函数(甲: 词频项, 乙: 词频项) -> 整数 {
        返回 乙.次数 - 甲.次数
    })
    返回 项们
}

函数 main() -> 整数 {
    私有 数组[字符串] 参数 = std.os.参数()
    如果 #参数 << 2 就 {
        std.io.写错误("用法：词频 <文件>")
        返回 1
    }

    私有 结果[字符串, 字符串] 读 = std.fs.读取文件(参数[1])
    匹配 读 {
        例 失败(原因): {
            std.io.写错误("读取失败：" .. 原因)
            返回 1
        }
        例 成功(文本): {
            私有 映射[字符串, 整数] 表 = 统计(借用 文本)
            私有 数组[词频项] 榜 = 排行(借用 表)

            私有 整数 前 = 10
            如果 #榜 << 前 就 { 前 = #榜 }
            对 i: 0~(前 - 1) {
                std.io.打印行(榜[i].词 .. "：" .. 字符串(榜[i].次数))
            }
            返回 0
        }
    }
}

函数 演示位运算() -> 整数 {
    私有 整数 甲 = 5 ^^ 3          -- 位异或，结果是 6
    私有 整数 乙 = ~~5             -- 按位取反，结果是 -6
    私有 整数 丙 = 5 <<< 2         -- 左移两位，结果是 20
    私有 整数 丁 = 20 >>> 1        -- 右移一位，结果是 10
    私有 整数 戊 = 7 // 2          -- 整除，结果是 3
    私有 整数 己 = 7 % 2           -- 取余，结果是 1
    私有 浮点 庚 = 2 ^ 10          -- 乘方，结果是 1024.0
    返回 甲 + 乙 + 丙 + 丁 + 戊 + 己
}
